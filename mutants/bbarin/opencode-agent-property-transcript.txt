import { describe, expect, test } from "bun:test"
import { FastCheck } from "effect/testing"
import { filterSkills } from "../../src/component/dialog-skill"

type SkillOption = { title: string; description?: string }

// Terms use a tiny alphabet so the filler characters below can be guaranteed
// absent from the term. "-" is the gap character used for character-level hits.
const term = FastCheck.array(FastCheck.constantFrom("a", "b", "c", "d", "e"), {
  minLength: 2,
  maxLength: 4,
}).map((letters) => letters.join(""))

// Four skills, one per R7 tier, built from the same query term:
//   tier 1 (keyword in the name): the term is contiguous in the title.
//   tier 2 (keyword in the description): the term is contiguous in the description only.
//   tier 3 (letters in the name): the term's characters are in order in the title with gaps.
//   tier 4 (letters in the description): the term's characters are in order in the description with gaps.
const tiers = (query: string): SkillOption[] => [
  { title: `name-${query}`, description: "zzzz" },
  { title: "1zzzz", description: `contains ${query} keyword` },
  { title: `2${query.split("").join("-")}`, description: "zzzz" },
  { title: "3zzzz", description: query.split("").join("-") },
]

// Random sort keys put the four tiers in a different input order on each run,
// so a filter that merely preserved input order would fail.
const order = FastCheck.array(FastCheck.integer(), { minLength: 4, maxLength: 4 })

describe("dialog skill filter (R7 property)", () => {
  test("orders keyword matches before character matches, and name before description", () => {
    FastCheck.assert(
      FastCheck.property(term, order, (query, keys) => {
        const expected = tiers(query)
        const shuffled = expected
          .map((skill, index) => ({ skill, key: keys[index] }))
          .sort((a, b) => a.key - b.key)
          .map((entry) => entry.skill)

        const actual = filterSkills(shuffled, query)

        expect(actual.map((skill) => skill.title)).toEqual(expected.map((skill) => skill.title))
      }),
      { numRuns: 200 },
    )
  })
})
